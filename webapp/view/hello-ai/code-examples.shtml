💻 Code Examples

These listings are the Hello AI demo. Copy the code into your own site under site/webapp/. Do not import this plugin's modules.

🎨 View: index.shtml

The page owns the textarea the AI agent operates on.

  • panel.create creates the chat panel on this tab.
  • /help lists every example; three of them are shown here.
  • /pad reports the pad size, and a pad-stats region shows the same count under the compose box.
  • A hidden /padreset command restores the demo text.
  • adapter.toolData supplies the textarea and the selection to read_draft and propose_draft_rewrite.
  • applyProposal replaces the pad and keeps the previous text for Undo in this tab.
  • executeTool appends text for append_draft.
// plugins/hello-ai/webapp/view/hello-ai/index.shtml

const draft = document.getElementById('hello-ai-draft');

function selectionOf() {
    const start = draft.selectionStart || 0;
    const end = draft.selectionEnd || 0;
    if (end <= start) {
        return '';
    }
    return draft.value.slice(start, end);
}

const panel = jPulse.ai.panel.create({
    scopeType: 'hello-ai',
    scopeId: 'demo',
    launcher: '#hello-ai-launch',
    examples: [
        '[[Read the scratch pad]]',
        '[[Shorten it]] — propose a shorter rewrite of the scratch pad',
        '[[What time is it on the server?]]'
    ],
    commands: [
        ...jPulse.ai.commands.defaults,
        {
            name: 'pad',
            hint: 'Scratch pad size and selection',
            run: function () {
                const start = draft.selectionStart || 0;
                const end = draft.selectionEnd || 0;
                const selected = end > start ? end - start : 0;
                return 'Scratch pad has ' + draft.value.length
                    + ' characters, ' + selected + ' selected';
            }
        }
    ],
    regions: [
        {
            name: 'pad-stats',
            anchor: 'composeBelow',
            render: function () {
                const start = draft.selectionStart || 0;
                const end = draft.selectionEnd || 0;
                const selected = end > start ? end - start : 0;
                return 'Scratch pad has ' + draft.value.length
                    + ' characters, ' + selected + ' selected';
            }
        }
    ],
    adapter: {
        _undoById: {},
        toolData: function (name) {
            if (name === 'read_draft' || name === 'propose_draft_rewrite') {
                return { text: draft.value, selection: selectionOf() };
            }
            return {};
        },
        renderProposalPreview: function (proposal) {
            return (proposal && proposal.preview) || '';
        },
        applyProposal: function (proposal) {
            const payload = (proposal && proposal.payload) || {};
            this._undoById[proposal.id] = draft.value;
            draft.value = String(payload.text || '');
            return true;
        },
        canUndoProposal: function (proposal) {
            return !!(proposal && this._undoById[proposal.id] != null);
        },
        undoProposal: function (proposal) {
            if (!this.canUndoProposal(proposal)) {
                return false;
            }
            draft.value = this._undoById[proposal.id];
            delete this._undoById[proposal.id];
            return true;
        },
        executeTool: function (name, args) {
            if (name !== 'append_draft') {
                return {
                    ok: false,
                    code: 'AI_UNKNOWN_TOOL',
                    error: 'Unknown client tool ' + name
                };
            }
            const text = String(args && args.text || '');
            const prefix = draft.value && !draft.value.endsWith('\n') ? '\n' : '';
            draft.value = draft.value + prefix + text;
            return {
                ok: true,
                data: { appended: text.length, totalChars: draft.value.length },
                summary: 'append_draft ' + text.length + ' chars'
            };
        },
        describeContext: function () {
            return 'Unsaved scratch pad';
        },
        describeTarget: function () {
            return selectionOf() || 'all text';
        }
    }
});

🎮 Controller: helloAi.js

The controller registers the tools.

  • Tools are pushed only when scopeType is hello-ai, so the rest of the site does not offer them.
  • read_draft, propose_draft_rewrite, and append_draft run in the browser.
  • get_hello_clock runs on the server. It returns the server time and the current username.
  • onAiScopeResolve sets the scope label. onAiToolData defaults module data. onAiPromptFragment adds the scratch-pad instructions.
  • Each tool in the file has a schema. Budgets include the over-limit message and hint.
// plugins/hello-ai/webapp/controller/helloAi.js

const SCOPE_TYPE = 'hello-ai';

function isHelloScope(ctx) {
    return (ctx?.actor?.scopeType || ctx?.scopeType) === SCOPE_TYPE;
}

class HelloAiController {
    static hooks = {
        onAiScopeResolve: { handler: 'onAiScopeResolve' },
        onAiToolRegister: { handler: 'onAiToolRegister' },
        onAiToolExecute: { handler: 'onAiToolExecute' },
        onAiToolData: { handler: 'onAiToolData' },
        onAiPromptFragment: { handler: 'onAiPromptFragment' }
    };

    static async onAiToolRegister(ctx) {
        if (!isHelloScope(ctx)) {
            return ctx;
        }
        // Each tool in the file also has a schema. Budgets include overMessage and overHint.
        ctx.tools.push(
            {
                name: 'read_draft',
                description: 'Read the scratch pad. data.text is the full text (capped at 32KB). data.excerpt is a 160-character preview.',
                host: 'client',
                module: 'readDraft',
                requires: 'scope:read'
            },
            {
                name: 'propose_draft_rewrite',
                description: 'Propose a full replacement of the scratch pad. Creates an Apply card; does not write.',
                host: 'client',
                module: 'proposeRewrite',
                requires: 'scope:write',
                proposes: true,
                dedupeArgs: true,
                budget: { key: 'proposals', max: 3 }
            },
            {
                name: 'append_draft',
                description: 'Append text to the scratch pad. Writes immediately; this is not a proposal.',
                host: 'client',
                requires: 'scope:write',
                mutates: true,
                budget: { key: 'writes', max: 3 }
            },
            {
                name: 'get_hello_clock',
                description: 'Return the server time and the current username.',
                host: 'server'
            }
        );
        return ctx;
    }

    static async onAiToolExecute(ctx) {
        if (ctx.tool?.name !== 'get_hello_clock') {
            return ctx;
        }
        ctx.result = {
            ok: true,
            data: {
                serverTime: new Date().toISOString(),
                username: ctx.actor?.username || ''
            },
            summary: 'get_hello_clock'
        };
        return ctx;
    }
}

📊 Module: readDraft.js

read_draft names this module. It reads the scratch pad and returns a result.

  • Takes text and selection from adapter.toolData.
  • Returns character, word, and line counts, plus the selection.
  • Returns the text, capped at 32 KB, and a 160-character excerpt.
  • Leaves the textarea unchanged.
// plugins/hello-ai/webapp/utils/ai-tools/readDraft.js

const EXCERPT_CHARS = 160;
const TEXT_CHARS = 32 * 1024;

export function run(data = {}, args = {}) {
    const text = String(data.text || '');
    const selection = String(data.selection || '');
    const excerptChars = Number.isFinite(args.excerptChars) ? args.excerptChars : EXCERPT_CHARS;
    const words = text.trim() ? text.trim().split(/\s+/).length : 0;
    const lines = text ? text.split(/\n/).length : 0;
    const excerpt = text.length > excerptChars ? `${text.slice(0, excerptChars).trim()}…` : text;
    const clipped = text.length > TEXT_CHARS;
    return {
        ok: true,
        data: {
            chars: text.length,
            words,
            lines,
            selection,
            excerpt,
            text: clipped ? text.slice(0, TEXT_CHARS) : text,
            truncated: clipped
        },
        summary: `read_draft ${words} words`
    };
}

📊 Module: proposeRewrite.js

propose_draft_rewrite names this module. A successful run returns a proposal. Apply writes later, in adapter.applyProposal on the view.

  • Requires a non-empty replacement text.
  • Rejects text longer than 32 KB.
  • Returns data.proposal with payload.text and the current length.
  • previewLine builds the one-line preview on the Apply card and clips each side to 80 characters.
  • Leaves the textarea unchanged.
// plugins/hello-ai/webapp/utils/ai-tools/proposeRewrite.js

const TEXT_CHARS = 32 * 1024;

export function run(data = {}, args = {}) {
    const current = String(data.text || '');
    const text = String(args.text || '');
    if (!text.trim()) {
        return {
            ok: false,
            code: 'AI_BAD_ARGS',
            error: 'text is required.',
            hint: 'Pass the full replacement text for the scratch pad.'
        };
    }
    if (text.length > TEXT_CHARS) {
        return {
            ok: false,
            code: 'AI_PROPOSAL_TOO_LARGE',
            error: `Proposed text is ${text.length} characters; the cap is ${TEXT_CHARS}.`
        };
    }
    return {
        ok: true,
        data: {
            proposal: {
                kind: 'rewrite',
                preview: previewLine(current, text),
                payload: {
                    text,
                    expectedChars: current.length
                }
            }
        },
        summary: `propose_draft_rewrite ${text.length} chars`
    };
}

🔗 Key Patterns

  • Scope gate — tools register only when scopeType is hello-ai
  • Client host — read_draft, propose_draft_rewrite, and append_draft run in this tab. A turn on this page uses the WebSocket
  • Server host — get_hello_clock runs in onAiToolExecute
  • Pure module — readDraft and proposeRewrite take data and args and return a result. They do not write the pad
  • Direct write — append_draft is mutates: true and has no module. adapter.executeTool appends immediately
  • Proposal — proposes: true shows an Apply card. Undo uses the snapshot stored on the adapter, in this tab only
  • Copy, don't import — a site writes the same shapes under site/webapp/

The Architecture page follows one read, one proposal, one append, and the server clock from the button to the textarea.
